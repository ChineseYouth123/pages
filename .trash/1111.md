<%*
// 获取用户输入
const category = await tp.system.prompt("请输入分类（音乐/视频等）");
const tag = await tp.system.prompt("请输入标签");
const content = await tp.system.prompt("请输入内容");

// 生成序号（需要读取当前表格的最大序号）
let maxNum = 0;
const activeFile = app.workspace.getActiveFile();
const fileContent = await app.vault.read(activeFile);
const lines = fileContent.split('\n');

for (const line of lines) {
    const match = line.match(/^\|\s*(\d+)\s*\|/);
    if (match && parseInt(match[1]) > maxNum) {
        maxNum = parseInt(match[1]);
    }
}

const newNum = maxNum + 1;

// 生成新行
const newRow = `| ${newNum} | ${tag} | ${category} | ${content} |`;

// 插入到表格最后一行之前
const insertIndex = lines.findIndex(line => line.includes('---')) + 2;
if (insertIndex > 0) {
    lines.splice(insertIndex, 0, newRow);
    await app.vault.modify(activeFile, lines.join('\n'));
} else {
    // 如果找不到分隔线，追加到末尾
    await app.vault.append(activeFile, `\n${newRow}`);
}

tp.file.cursor(0);
%>
